"""Deterministic C2.1 interpretation of compositional business semantics.

The interpreter consumes only a normalized header and a validated governed
catalog. It emits evidence and optional exact legacy projections; it does not
read XLSX files, call an LLM, calculate values, or authorize downstream use.
"""

from __future__ import annotations

import re
import unicodedata
from pathlib import Path
from typing import Any, Final, Sequence

from pymia.smartpyme.service_1_compositional_business_semantics_contract_v1 import (
    AMOUNT_SCOPE_ENTITY_PERIOD,
    AMOUNT_SCOPE_UNIT,
    AMOUNT_SCOPE_UNKNOWN,
    GRAIN_HINT_ENTITY_PERIOD,
    Service1CompositionalBusinessSemanticV1,
    Service1CompositionalSemanticCatalogV1,
    build_service_1_compositional_business_semantic_v1,
    load_service_1_compositional_semantic_catalog_v1,
)


SCHEMA_VERSION: Final[str] = "SERVICE_1_COMPOSITIONAL_BUSINESS_SEMANTICS_V1"
_NON_ALNUM_RE: Final[re.Pattern[str]] = re.compile(r"[^a-z0-9_]+")
_UNDERSCORE_RE: Final[re.Pattern[str]] = re.compile(r"_+")


def normalize_service_1_compositional_header_v1(value: Any) -> str:
    text = unicodedata.normalize("NFKD", str(value or "")).encode("ascii", "ignore").decode("ascii")
    text = text.casefold().strip().replace("-", "_").replace("/", "_")
    text = _NON_ALNUM_RE.sub("_", text)
    return _UNDERSCORE_RE.sub("_", text).strip("_")


def _catalog(
    semantic_catalog: Service1CompositionalSemanticCatalogV1 | None,
    catalog_path: str | Path | None,
) -> Service1CompositionalSemanticCatalogV1:
    if semantic_catalog is not None and catalog_path is not None:
        raise ValueError("provide semantic_catalog or catalog_path, not both")
    return semantic_catalog or load_service_1_compositional_semantic_catalog_v1(catalog_path)


def infer_service_1_compositional_business_semantic_v1(
    column_name: str,
    *,
    normalized_header: str | None = None,
    sample_values: Sequence[Any] = (),
    sheet_name: str | None = None,
    co_column_names: Sequence[str] = (),
    inferred_data_type: str | None = None,
    semantic_catalog: Service1CompositionalSemanticCatalogV1 | None = None,
    catalog_path: str | Path | None = None,
) -> Service1CompositionalBusinessSemanticV1 | None:
    """Infer business evidence only when the governed catalog proves both axes."""
    del sample_values, sheet_name, co_column_names, inferred_data_type
    catalog = _catalog(semantic_catalog, catalog_path)
    normalized = normalize_service_1_compositional_header_v1(normalized_header or column_name)
    measure_kind, measure_token = catalog.measure_for_header(normalized)
    concept, concept_token = catalog.concept_for_header(normalized)
    if measure_kind is None or concept is None:
        return None
    amount_scope, grain_hint, scope_token = catalog.scope_for_header(normalized)
    role, variable = catalog.runtime_projection(
        measure_kind=measure_kind,
        business_concept=concept,
        amount_scope=amount_scope,
        matched_alias=concept_token,
    )
    evidence = [
        f"normalized_header: '{normalized}'",
        f"measure_token: '{measure_token}'",
        f"business_concept_token: '{concept_token}'",
        f"amount_scope: '{amount_scope}'",
    ]
    if scope_token is not None:
        evidence.append(f"scope_token: '{scope_token}'")
    if role is None:
        evidence.append("no_exact_runtime_projection")
    return build_service_1_compositional_business_semantic_v1(
        measure_kind=measure_kind,
        business_concept=concept,
        amount_scope=amount_scope,
        semantic_grain_hint=grain_hint or GRAIN_HINT_ENTITY_PERIOD,
        confidence=0.95,
        evidence=evidence,
        runtime_semantic_role=role,
        runtime_variable_name=variable,
        allowed_business_concepts=catalog.business_concept_ids,
    )


def build_service_1_compositional_business_semantics_from_headers_v1(
    headers: Sequence[str],
    *,
    sheet_name: str | None = None,
    semantic_catalog: Service1CompositionalSemanticCatalogV1 | None = None,
    catalog_path: str | Path | None = None,
) -> dict[str, Service1CompositionalBusinessSemanticV1]:
    catalog = _catalog(semantic_catalog, catalog_path)
    result: dict[str, Service1CompositionalBusinessSemanticV1] = {}
    for header in headers:
        descriptor = infer_service_1_compositional_business_semantic_v1(
            header,
            sheet_name=sheet_name,
            co_column_names=headers,
            semantic_catalog=catalog,
        )
        if descriptor is not None:
            result[str(header)] = descriptor
    return result


def render_service_1_compositional_owner_proposal_v1(
    column_name: str,
    descriptor: Service1CompositionalBusinessSemanticV1,
    *,
    semantic_catalog: Service1CompositionalSemanticCatalogV1 | None = None,
    catalog_path: str | Path | None = None,
) -> str:
    if not isinstance(descriptor, Service1CompositionalBusinessSemanticV1):
        raise ValueError("descriptor must be a Service1CompositionalBusinessSemanticV1")
    catalog = _catalog(semantic_catalog, catalog_path)
    if descriptor.runtime_semantic_role == "unit_cost_candidate":
        meaning = "el costo por unidad"
    else:
        measure = catalog.label_for_measure(descriptor.measure_kind)
        concept = catalog.label_for_concept(descriptor.business_concept)
        if descriptor.amount_scope == AMOUNT_SCOPE_ENTITY_PERIOD:
            suffix = " de cada entidad/período"
        elif descriptor.amount_scope == "period_total_amount":
            suffix = " total del período"
        elif descriptor.amount_scope == "line_amount":
            suffix = " de cada línea"
        elif descriptor.amount_scope == AMOUNT_SCOPE_UNKNOWN:
            suffix = " (alcance del importe aún no resuelto)"
        elif descriptor.amount_scope == AMOUNT_SCOPE_UNIT:
            suffix = " por unidad"
        else:  # pragma: no cover - contract currently enumerates all scopes
            suffix = " (alcance del importe aún no resuelto)"
        # A governed catalog may intentionally keep a concept label generic
        # until the owner gives it a more precise business name.  In that
        # case, repeating the generic noun after the measure produces awkward
        # language (for example, "importe cobrado de un importe del
        # negocio").  Preserve the evidence while leaving the unresolved
        # concept explicit through the measure/scope wording.
        if concept.casefold().startswith(("un ", "una ")):
            meaning = f"{measure}{suffix}"
        else:
            meaning = f"{measure} de {concept}{suffix}"
    return f"Entendí que '{column_name}' representa {meaning}. ¿Está bien?"


# Short aliases keep one implementation discoverable under the names used by
# the C2 design notes; they do not create a second semantic engine.
infer_service_1_compositional_semantic_v1 = infer_service_1_compositional_business_semantic_v1
build_service_1_compositional_semantics_from_headers_v1 = build_service_1_compositional_business_semantics_from_headers_v1
